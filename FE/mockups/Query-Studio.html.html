-- Find pending orders with user details
SELECT
  u.name,
  u.email,
  o.total,
  o.status
FROM orders o
JOIN users u ON u.id = o.user_id
WHERE o.status = 'pending'
ORDER BY o.created_at DESC
LIMIT 100;
